/**
 * 可选的目录轮询刷新（PRD FR-3.9 P2 / 开发计划 T-4.3）。
 *
 * WebDAV 协议无变更推送能力，`watch()` 只能是 no-op。轮询是唯一的近似手段，
 * 但代价真实：每轮都是一次 PROPFIND，目录多时会持续消耗服务端资源。
 * 因此**默认关闭**，且只轮询「当前可见」的目录——用户看不到的地方刷新它没有意义。
 *
 * 与 FR-2.4 手动刷新的关系：轮询是锦上添花，手动刷新才是保底手段。
 *
 * 不依赖 `vscode`：变更通知通过 onChanged 回调上抛，由 fs/provider.ts 转成
 * FileChangeEvent。这样指纹逻辑可脱离扩展宿主单测（NFR-4.1）。
 */
import type { MetadataCache } from './cache.ts';
import type { ConnectionResolver } from '../connection/resolver.ts';
import type { DirEntry } from '../webdav/types.ts';
 
export interface PollerDeps {
  cache: MetadataCache;
  resolver: ConnectionResolver;
  /** 检测到变化时通知 VSCode 刷新。 */
  onChanged: (connectionId: string, path: string) => void;
  /** 日志由上层注入——直接 import log/channel.ts 会把 vscode 依赖带进本模块。 */
  log?: { info(msg: string): void; debug(msg: string): void };
}
 
export class DirectoryPoller {
  private readonly deps: PollerDeps;
  private timer: ReturnType<typeof setInterval> | undefined;
  private intervalMs = 0;
 
  /** 正在轮询的目录：`connectionIdpath` → 上次的目录指纹。 */
  private readonly watched = new Map<string, string>();
 
  /** 防止上一轮未结束就开下一轮（慢服务端上会堆积）。 */
  private ticking = false;
 
  constructor(deps: PollerDeps) {
    this.deps = deps;
  }
 
  dispose(): void {
    this.stop();
    this.watched.clear();
  }
 
  /** 按配置启停。intervalSeconds <= 0 表示关闭。 */
  configure(intervalSeconds: number): void {
    const ms = Math.max(0, intervalSeconds) * 1000;
    if (ms === this.intervalMs) return;
    this.intervalMs = ms;
    this.stop();
    if (ms > 0) {
      this.deps.log?.info(`目录轮询已启用，间隔 ${intervalSeconds}s（轮询会持续产生 PROPFIND 请求）`);
      this.timer = setInterval(() => void this.tick(), ms);
    } else {
      this.deps.log?.info('目录轮询已关闭');
    }
  }
 
  private stop(): void {
    if (this.timer) clearInterval(this.timer);
    this.timer = undefined;
  }
 
  /** 登记一个需要关注的目录（通常是当前打开的工作区目录）。 */
  track(connectionId: string, path: string, entries: DirEntry[]): void {
    if (this.intervalMs === 0) return;
    this.watched.set(key(connectionId, path), fingerprint(entries));
  }
 
  untrack(connectionId: string, path: string): void {
    this.watched.delete(key(connectionId, path));
  }
 
  private async tick(): Promise<void> {
    if (this.ticking || this.watched.size === 0) return;
    this.ticking = true;
    try {
      for (const [k, previous] of [...this.watched]) {
        const [connectionId, path] = k.split('') as [string, string];
        try {
          const { dav } = this.deps.resolver.resolve(connectionId);
          const entries = await dav.list(path, '轮询刷新');
          const current = fingerprint(entries);
          if (current !== previous) {
            this.deps.log?.debug(`轮询发现变化: ${connectionId}${path}`);
            this.watched.set(k, current);
            this.deps.cache.setChildren(connectionId, path, entries);
            this.deps.onChanged(connectionId, path);
          }
        } catch {
          // 轮询失败不打扰用户：手动刷新与实际操作仍会给出明确错误
          this.watched.delete(k);
        }
      }
    } finally {
      this.ticking = false;
    }
  }
}
 
function key(connectionId: string, path: string): string {
  return `${connectionId}${path}`;
}
 
/**
 * 目录指纹：名称 + 大小 + mtime。
 * 只比条目数会漏掉「改了内容但没增删文件」这一最常见的场景。
 */
export function fingerprint(entries: DirEntry[]): string {
  return entries
    .map((e) => `${e.name}:${e.isDirectory ? 'd' : 'f'}:${e.stat.size}:${e.stat.mtime}`)
    .sort()
    .join('|');
}
